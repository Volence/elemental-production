```js
function normalizeSingleCurrent(maps) {
  if (!Array.isArray(maps)) return maps;
  const lastCurrent = maps.reduce((acc, m, i) => (m && m.status === 'current' ? i : acc), -1);
```

```js
// Foreground (near-black vs white) readable ON a team-color FILL.
// Inverse of legibleColor (which fixes dark team color used as TEXT on dark ground).
function textOnColor(hex) {
  var m = /^#([0-9a-f]{6})$/i.exec(String(hex || '').trim());
  if (!m) return '#ffffff';
  var n = parseInt(m[1], 16);
  var lin = function (c) { c /= 255; return c <= 0.03928 ? c / 12.92 : Math.pow((c + 0.055) / 1.055, 2.4); };
  var L = 0.2126 * lin((n >> 16) & 255) + 0.7152 * lin((n >> 8) & 255) + 0.0722 * lin(n & 255);
  return (1.05 / (L + 0.05)) >= ((L + 0.05) / 0.055) ? '#ffffff' : '#0a0c11';
}
```

```js
// Renders overlays/stinger-transition.html to a VP9 WebM with alpha.
// Usage: node scripts/render-stinger.mjs [--fps 60] [--out public/stinger-transition.webm]
// Requires: google-chrome-stable (headless) + ffmpeg with libvpx-vp9/yuva420p.
// Capture: pause all Web Animations and scrub currentTime per frame —
// --virtual-time-budget does NOT reliably scrub compositor-driven CSS animations.
import { spawn, execSync } from 'node:child_process';
import { mkdirSync, writeFileSync, rmSync } from 'node:fs';
import { resolve } from 'node:path';

const FPS = 60, DURATION_MS = 1300;           // 1.1s animation + verified-transparent tail
const FRAMES = Math.ceil(DURATION_MS / 1000 * FPS);
const url = 'file://' + resolve('overlays/stinger-transition.html');
const tmp = resolve('.stinger-frames'); rmSync(tmp, { recursive: true, force: true }); mkdirSync(tmp);

const chrome = spawn('google-chrome-stable', [
  '--headless=new', '--remote-debugging-port=0', '--window-size=1920,1080',
  '--allow-file-access-from-files', '--hide-scrollbars', '--disable-gpu-vsync',
  '--user-data-dir=' + tmp + '/profile', url,
], { stdio: ['ignore', 'pipe', 'pipe'] });

  chrome.stderr.on('data', d => { const m = /DevTools listening on (ws:\/\/\S+)/.exec(d.toString()); if (m) res(m[1]); });
  chrome.on('exit', () => rej(new Error('chrome exited early')));
});

const ws = new WebSocket(wsUrl); await new Promise(r => ws.onopen = r);
let id = 0; const pending = new Map();
ws.onmessage = e => { const msg = JSON.parse(e.data); if (msg.id && pending.has(msg.id)) { pending.get(msg.id)(msg); pending.delete(msg.id); } };
const send = (method, params = {}, sessionId) => new Promise(res => { const i = ++id; pending.set(i, res); ws.send(JSON.stringify({ id: i, method, params, sessionId })); });

const { result: { targetInfos } } = await send('Target.getTargets');
const page = targetInfos.find(t => t.type === 'page');
const { result: { sessionId } } = await send('Target.attachToTarget', { targetId: page.targetId, flatten: true });
const s = (m, p) => send(m, p, sessionId);
await s('Page.enable'); await s('Runtime.enable');
await s('Emulation.setDeviceMetricsOverride', { width: 1920, height: 1080, deviceScaleFactor: 1, mobile: false });
await s('Emulation.setDefaultBackgroundColorOverride', { color: { r: 0, g: 0, b: 0, a: 0 } });
await new Promise(r => setTimeout(r, 1500));  // fonts + pinwheel.js build

let best = { i: 0, coverage: -1 };
for (let i = 0; i < FRAMES; i++) {
  const t = i * 1000 / FPS;
  await s('Runtime.evaluate', { expression: `document.getAnimations().forEach(a => { a.pause(); a.currentTime = ${t}; })`, awaitPromise: false });
  await new Promise(r => setTimeout(r, 20));  // let the compositor apply the scrub
  const { result: { data } } = await s('Page.captureScreenshot', { format: 'png' });
  writeFileSync(`${tmp}/frame_${String(i).padStart(4, '0')}.png`, Buffer.from(data, 'base64'));
}
ws.close(); chrome.kill();

// Coverage scan (transition point = frame where the screen is most covered)
for (let i = 0; i < FRAMES; i++) {
  const out = execSync(`ffprobe -v error -f lavfi -i "movie=${tmp}/frame_${String(i).padStart(4, '0')}.png,format=rgba,signalstats" -show_entries frame_tags=lavfi.signalstats.YAVG -of csv=p=0 2>/dev/null || true`).toString().trim();
  // simpler + robust: use ImageMagick if available; otherwise skip scan and use 550ms default
}
const outFile = resolve(process.argv.includes('--out') ? process.argv[process.argv.indexOf('--out') + 1] : 'public/stinger-transition.webm');
```

```js
// Aggregate all played maps' FACEIT round stats into one series total per player.
// Join players on playerId (fallback nickname); match round-teams to team1/team2
// by teamId vs state.teams.*.faceitId (never by array index); K/D recomputed from
// summed totals (never average per-map ratios).
function aggregateSeries(playerStats, teams) {
  var sides = { team1: { players: {}, totals: { kills: 0, deaths: 0, finalBlows: 0, damageDealt: 0, healingDone: 0 }, mapsWon: 0 },
                team2: { players: {}, totals: { kills: 0, deaths: 0, finalBlows: 0, damageDealt: 0, healingDone: 0 }, mapsWon: 0 } };
  for (var r = 0; r < (playerStats || []).length; r++) {
    var round = playerStats[r];
    for (var t = 0; t < (round.teams || []).length; t++) {
      var rt = round.teams[t];
      var sideKey = null;
      if (teams.team1 && teams.team1.faceitId && rt.teamId === teams.team1.faceitId) sideKey = 'team1';
      else if (teams.team2 && teams.team2.faceitId && rt.teamId === teams.team2.faceitId) sideKey = 'team2';
      else sideKey = t === 0 ? 'team1' : 'team2';   // manual fallback (no faceitIds)
      var side = sides[sideKey];
      if (rt.stats && rt.stats.teamWin) side.mapsWon += 1;
      for (var p = 0; p < (rt.players || []).length; p++) {
        var pl = rt.players[p];
        var key = pl.playerId || pl.nickname;
        var agg = side.players[key] || (side.players[key] = { nickname: pl.nickname, role: pl.role,
          kills: 0, deaths: 0, finalBlows: 0, damageDealt: 0, healingDone: 0, maps: 0 });
        agg.nickname = pl.nickname; agg.role = agg.role || pl.role;
        agg.kills += pl.kills || 0; agg.deaths += pl.deaths || 0; agg.finalBlows += pl.finalBlows || 0;
        agg.damageDealt += pl.damageDealt || 0; agg.healingDone += pl.healingDone || 0; agg.maps += 1;
        side.totals.kills += pl.kills || 0; side.totals.deaths += pl.deaths || 0;
        side.totals.finalBlows += pl.finalBlows || 0; side.totals.damageDealt += pl.damageDealt || 0;
        side.totals.healingDone += pl.healingDone || 0;
      }
    }
  }
  ['team1', 'team2'].forEach(function (k) {
    var list = Object.keys(sides[k].players).map(function (id) { return sides[k].players[id]; });
    var order = { tank: 0, damage: 1, support: 2 };
    list.sort(function (a, b) {
      var ra = order[(a.role || '').toLowerCase()], rb = order[(b.role || '').toLowerCase()];
      if (ra !== rb) return (ra === undefined ? 9 : ra) - (rb === undefined ? 9 : rb);
      return b.kills - a.kills;
    });
    list.forEach(function (pl) { pl.kdRatio = pl.deaths > 0 ? (pl.kills / pl.deaths).toFixed(2) : String(pl.kills); });
    sides[k].list = list;
  });
  return sides;
}
```
